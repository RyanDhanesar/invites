  <script>
    /* 1. Bidirectional Parallax Scroll Observer */
    const revealObserver = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add('active');
        } else {
          entry.target.classList.remove('active');
        }
      });
    }, {
      threshold: 0.12,
      rootMargin: "0px 0px -40px 0px"
    });

    document.querySelectorAll('.reveal').forEach(el => revealObserver.observe(el));

    /* 2. Real-Time Countdown to February 12, 2027 */
    const targetDate = new Date("February 12, 2027 15:30:00").getTime();

    function updateCountdown() {
      const now = new Date().getTime();
      const distance = targetDate - now;

      if (distance > 0) {
        document.getElementById("cd-days").innerText = Math.floor(distance / (1000 * 60 * 60 * 24));
        document.getElementById("cd-hours").innerText = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
        document.getElementById("cd-mins").innerText = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
        document.getElementById("cd-secs").innerText = Math.floor((distance % (1000 * 60)) / 1000);
      }
    }
    setInterval(updateCountdown, 1000);
    updateCountdown();

    /* 3. Ambient Audio Player Simulation */
    let isPlaying = false;
    let trackInterval;

    function togglePlayMusic() {
      const icon = document.getElementById('playIcon');
      const bar = document.getElementById('trackProgress');
      isPlaying = !isPlaying;

      if (isPlaying) {
        icon.innerHTML = '<path d="M6 19h4V5H6v14zm8-14v14h4V5h-4z"/>';
        let progress = 38;
        trackInterval = setInterval(() => {
          progress = (progress + 1) % 100;
          bar.style.width = progress + '%';
        }, 600);
      } else {
        icon.innerHTML = '<path d="M8 5v14l11-7z"/>';
        clearInterval(trackInterval);
      }
    }

    /* 4. RSVP Modal Management */
    function openRSVPModal() {
      document.getElementById('rsvpModal').classList.add('active');
    }

    function closeRSVPModal() {
      document.getElementById('rsvpModal').classList.remove('active');
    }

    function submitRSVP(e) {
      e.preventDefault();
      alert("Thank you! Your attendance response has been received.");
      closeRSVPModal();
    }

    function openEnvelopeSequence() {
      const envelope = document.getElementById('envelopeWrapper');
      const overlay = document.getElementById('envelope-overlay');

      envelope.style.pointerEvents = 'none';

      // 1. All 4 flaps unfold outward in 3D right on top of the live page
      envelope.classList.add('open');

      // 2. Once flaps unfold flat, remove the overlay layer so the guest can scroll freely
      setTimeout(() => {
        overlay.classList.add('unwrapped');
        overlay.style.display = 'none';
      }, 1800);
    }
  </script>
